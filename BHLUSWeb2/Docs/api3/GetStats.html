GetStats
Get the numbers of Titles, Items, Pages, and Parts currently in BHL.
SYNTAX
https://www.biodiversitylibrary.org/api3?op=GetStats
&format=<"xml" for an XML response or "json" for JSON (OPTIONAL; "xml" is the default)>
&apikey=<API key value>
EXAMPLE REQUEST (XML)
            
 https://www.biodiversitylibrary.org/api3?op=GetStatss&format=xml&apikey=12345678-BBBB-DDDD-FFFF-123456789012
EXAMPLE RESPONSE (XML)
            

<?xml version="1.0" encoding="utf-8"?>
<Response
  xmlns:xsd="http://www.w3.org/2001/XMLSchema"
  xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance">
  <Status>ok</Status>
  <ErrorMessage />
  <Result>
    <TitleCount>130</TitleCount>
    <ItemCount>212</ItemCount>
    <PageCount>65279</PageCount>
    <PartCount>5588</PartCount>
  </Result>
</Response>
EXAMPLE REQUEST (JSON)
            
 https://www.biodiversitylibrary.org/api3?op=GetStats&format=json&apikey=12345678-BBBB-DDDD-FFFF-123456789012
EXAMPLE RESPONSE (JSON)
            
{
  "Status": "ok",
  "ErrorMessage": "",
  "Result": {
    "TitleCount": "130",
    "ItemCount": "212",
    "PageCount": "65279"
    "PartCount": "5588",
  }
}
